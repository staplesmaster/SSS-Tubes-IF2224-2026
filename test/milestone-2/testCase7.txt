program Test4;

type Point ==
record
    x: integer; 
    y: integer
end;

var
    now, ends: Point;
    i: integer;

begin
    now := Point(0, 0);
    ends := Point(10, 10);
    for i := 10 downto 0 do
        case i of 
            x : now.x := now.x + 1;
            y : now.y := now.y + 1;
        end;
    print('Point = ', x, 'And', y);
end.
